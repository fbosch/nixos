cmake_minimum_required(VERSION 3.30)
project(persistent-position LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
add_library(persistent-position SHARED main.cpp)
target_link_libraries(persistent-position PRIVATE PkgConfig::HYPRLAND)
include(CTest)
if(BUILD_TESTING)
    add_executable(persistent-position-test position_store_test.cpp)
    add_test(NAME persistent-position-test COMMAND persistent-position-test)
    add_executable(persistent-position-storage-test position_storage_test.cpp)
    add_test(NAME persistent-position-storage-test COMMAND persistent-position-storage-test)
endif()
